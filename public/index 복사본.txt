  <script>
    (() => {
      const STORAGE_KEYS = {
        API_KEY: 'botanical_note_gemini_api_key_v1',
        SETTINGS: 'botanical_note_settings_v1'
      };
      const DB_NAME = 'BotanicalNoteDB';
      const DB_VERSION = 1;
      const STORE_NAME = 'entries';

      const state = {
        view: 'browse',
        previewUrl: '',
        previewBase64: '',
        previewMimeType: 'image/jpeg',
        analysisResult: null,
        referencePhotos: [],
        entries: [],
        selectedEntryId: null,
        deleteMode: false,
        selectedIds: new Set(),
        settings: loadSettings()
      };

      function loadSettings() {
        try {
          const raw = localStorage.getItem(STORAGE_KEYS.SETTINGS);
          if (raw) {
            const parsed = JSON.parse(raw);
            return {
              recordDate: parsed.recordDate !== false,
              recordLocation: parsed.recordLocation !== false
            };
          }
        } catch (_) {}
        return { recordDate: true, recordLocation: true };
      }

      function saveSettings(next) {
        state.settings = { ...next };
        localStorage.setItem(STORAGE_KEYS.SETTINGS, JSON.stringify(state.settings));
      }

      function getApiKey() {
        return (localStorage.getItem(STORAGE_KEYS.API_KEY) || '').trim();
      }

      function setApiKey(key) {
        localStorage.setItem(STORAGE_KEYS.API_KEY, key.trim());
      }

      function openDB() {
        return new Promise((resolve, reject) => {
          const req = indexedDB.open(DB_NAME, DB_VERSION);
          req.onupgradeneeded = (e) => {
            const db = e.target.result;
            if (!db.objectStoreNames.contains(STORE_NAME)) {
              const store = db.createObjectStore(STORE_NAME, { keyPath: 'id' });
              store.createIndex('createdAt', 'createdAt', { unique: false });
            }
          };
          req.onsuccess = () => resolve(req.result);
          req.onerror = () => reject(req.error);
        });
      }

      async function fetchAllEntries() {
        const db = await openDB();
        return new Promise((resolve, reject) => {
          const tx = db.transaction(STORE_NAME, 'readonly');
          const store = tx.objectStore(STORE_NAME);
          const req = store.getAll();
          req.onsuccess = () => {
            const list = req.result || [];
            list.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
            resolve(list);
          };
          req.onerror = () => reject(req.error);
        });
      }

      async function saveEntryToDB(entry) {
        const db = await openDB();
        return new Promise((resolve, reject) => {
          const tx = db.transaction(STORE_NAME, 'readwrite');
          tx.objectStore(STORE_NAME).put(entry);
          tx.oncomplete = () => resolve();
          tx.onerror = () => reject(tx.error);
        });
      }

      async function deleteEntriesFromDB(ids) {
        const db = await openDB();
        return new Promise((resolve, reject) => {
          const tx = db.transaction(STORE_NAME, 'readwrite');
          const store = tx.objectStore(STORE_NAME);
          ids.forEach((id) => store.delete(id));
          tx.oncomplete = () => resolve();
          tx.onerror = () => reject(tx.error);
        });
      }

      let audioCtx = null;
      function ensureAudioContext() {
        const Ctx = window.AudioContext || window.webkitAudioContext;
        if (!Ctx) return null;
        if (!audioCtx) audioCtx = new Ctx();
        if (audioCtx.state === 'suspended') audioCtx.resume().catch(() => {});
        return audioCtx;
      }

      function playLeafRustleSound() {
        const ctx = ensureAudioContext();
        if (!ctx) return;
        try {
          const duration = 0.85;
          const sampleRate = ctx.sampleRate;
          const frameCount = Math.floor(sampleRate * duration);
          const buffer = ctx.createBuffer(1, frameCount, sampleRate);
          const data = buffer.getChannelData(0);
          for (let i = 0; i < frameCount; i++) {
            const t = i / sampleRate;
            const pulse1 = Math.pow(Math.max(0, Math.sin(t * Math.PI * 7)), 4);
            const pulse2 = Math.pow(Math.max(0, Math.sin(t * Math.PI * 11 + 0.8)), 5);
            const fadeOut = 1 - t / duration;
            const crinkle = (Math.random() * 2 - 1) * (0.25 + 0.75 * Math.random());
            data[i] = crinkle * (0.25 + 0.75 * Math.max(pulse1, pulse2)) * fadeOut;
          }
          const noise = ctx.createBufferSource();
          noise.buffer = buffer;
          const bandpass = ctx.createBiquadFilter();
          bandpass.type = 'bandpass';
          bandpass.frequency.setValueAtTime(2900, ctx.currentTime);
          bandpass.Q.setValueAtTime(1.1, ctx.currentTime);
          const highpass = ctx.createBiquadFilter();
          highpass.type = 'highpass';
          highpass.frequency.setValueAtTime(950, ctx.currentTime);
          const gain = ctx.createGain();
          gain.gain.setValueAtTime(0.001, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.32, ctx.currentTime + 0.05);
          gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
          noise.connect(bandpass);
          bandpass.connect(highpass);
          highpass.connect(gain);
          gain.connect(ctx.destination);
          noise.start();
          noise.stop(ctx.currentTime + duration);
        } catch (_) {}
      }

      function getCurrentLocationText() {
        return new Promise((resolve) => {
          if (!navigator.geolocation) {
            resolve('위치 정보 미지원 기기');
            return;
          }
          navigator.geolocation.getCurrentPosition(
            (pos) => resolve(`위도 ${pos.coords.latitude.toFixed(4)}, 경도 ${pos.coords.longitude.toFixed(4)}`),
            () => resolve('위치 권한 꺼짐 또는 확인 불가'),
            { enableHighAccuracy: true, timeout: 6000, maximumAge: 60000 }
          );
        });
      }

      function readImageFile(file) {
        return new Promise((resolve, reject) => {
          const reader = new FileReader();
          reader.onload = () => {
            const img = new Image();
            img.onload = () => {
              const maxSide = 768;
              let { width, height } = img;
              if (width > maxSide || height > maxSide) {
                if (width > height) {
                  height = Math.round((height * maxSide) / width);
                  width = maxSide;
                } else {
                  width = Math.round((width * maxSide) / height);
                  height = maxSide;
                }
              }
              const canvas = document.createElement('canvas');
              canvas.width = width;
              canvas.height = height;
              const ctx = canvas.getContext('2d');
              ctx.drawImage(img, 0, 0, width, height);
              const dataUrl = canvas.toDataURL('image/jpeg', 0.78);
              const base64 = dataUrl.split(',')[1] || '';
              resolve({ dataUrl, base64, mimeType: 'image/jpeg' });
            };
            img.onerror = reject;
            img.src = reader.result;
          };
          reader.onerror = reject;
          reader.readAsDataURL(file);
        });
      }

      async function identifyPlantWithGemini(apiKey, base64Data, mimeType) {
        const models = ["gemini-3.5-flash-lite", "gemini-3.8-flash", "gemini-4-hydrogen"];
        const loadingText = document.getElementById("loadingText");
        const promptText = [
          "당신은 식물학 백과사전 전문가입니다. 첨부된 사진 속 식물을 판별하고 아래 JSON 스키마로만 응답하세요.",
          "- name_ko: 한국어 일반 식물 이름",
          "- scientific_name: 라틴어 학명",
          "- english_name: 영어 일반명 또는 검색용 영문명",
          "- description: 잎, 꽃, 줄기/서식지, 비슷한 식물과의 구별 포인트를 백과사전식으로 알기 쉽게 정리한 한국어 설명"
        ].join("\n");

        const payload = JSON.stringify({
          contents: [{ parts: [{ text: promptText }, { inline_data: { mime_type: mimeType, data: base64Data } }] }],
          generationConfig: {
            responseMimeType: "application/json",
            responseSchema: {
              type: "OBJECT",
              properties: {
                name_ko: { type: "STRING" },
                scientific_name: { type: "STRING" },
                english_name: { type: "STRING" },
                description: { type: "STRING" }
              },
              required: ["name_ko", "scientific_name", "english_name", "description"]
            }
          }
        });

        let lastError = null;
        const delays = [900, 1600, 2600, 4000, 6000];

        for (let round = 0; round < 2; round++) {
          for (const model of models) {
            const endpoint = "https://generativelanguage.googleapis.com/v1beta/models/" + model + ":generateContent?key=" + encodeURIComponent(apiKey);

            for (let attempt = 0; attempt <= delays.length; attempt++) {
              if (loadingText) {
                loadingText.textContent = attempt === 0 && round === 0
                  ? ("식물 특징을 살펴보는 중... (" + model + ")")
                  : ("서버 응답을 다시 요청하는 중... (" + model + ")");
              }

              try {
                const response = await fetch(endpoint, {
                  method: "POST",
                  headers: { "Content-Type": "application/json" },
                  body: payload
                });

                if (response.ok) {
                  const data = await response.json();
                  const rawText = data?.candidates?.[0]?.content?.parts?.[0]?.text || "";
                  const cleaned = rawText.replace(/^```json\s*/i, "").replace(/```$/i, "").trim();
                  const parsed = JSON.parse(cleaned);
                  return {
                    nameKo: (parsed.name_ko || "이름 미확인 식물").trim(),
                    scientificName: (parsed.scientific_name || "Species indeterminata").trim(),
                    englishName: (parsed.english_name || parsed.scientific_name || "plant").trim(),
                    description: (parsed.description || "상세 특징 정보가 제공되지 않았습니다.").trim()
                  };
                }

                const errText = await response.text();
                lastError = new Error("Gemini API 오류 (" + response.status + " · " + model + "): " + errText.slice(0, 180));

                if (response.status === 503 || response.status === 429 || response.status === 500 || response.status === 502 || response.status === 504) {
                  const waitTime = delays[Math.min(attempt, delays.length - 1)] + Math.floor(Math.random() * 450);
                  await new Promise((r) => setTimeout(r, waitTime));
                  if (attempt >= 2) break;
                  continue;
                }

                if (response.status === 404) {
                  break;
                }

                throw lastError;
              } catch (netErr) {
                lastError = netErr;
                const waitTime = delays[Math.min(attempt, delays.length - 1)] + 250;
                await new Promise((r) => setTimeout(r, waitTime));
                if (attempt >= 2) break;
              }
            }
          }
        }

        throw lastError || new Error("잠시 후 다시 시도해 주세요.");
      }

      async function fetchReferencePhotos(scientificName, englishName, koreanName) {
        const queries = [scientificName, englishName, koreanName].filter(Boolean);
        const collected = [];
        const seenUrls = new Set();

        for (const q of queries) {
          if (collected.length >= 4) break;
          try {
            const url = `https://commons.wikimedia.org/w/api.php?action=query&format=json&origin=*&generator=search&gsrnamespace=6&gsrlimit=12&gsrsearch=${encodeURIComponent(q + ' filetype:bitmap')}&prop=imageinfo&iiprop=url|extmetadata&iiurlwidth=400`;
            const res = await fetch(url);
            if (!res.ok) continue;
            const json = await res.json();
            const pages = json?.query?.pages ? Object.values(json.query.pages) : [];
            for (const page of pages) {
              if (collected.length >= 4) break;
              const info = page?.imageinfo?.[0];
              const thumbUrl = info?.thumburl || info?.url;
              if (!thumbUrl) continue;
              const lower = thumbUrl.toLowerCase();
              if (lower.endsWith('.svg') || lower.endsWith('.tif') || lower.endsWith('.tiff')) continue;
              if (seenUrls.has(thumbUrl)) continue;
              seenUrls.add(thumbUrl);
              collected.push({ src: thumbUrl, title: (page.title || '').replace(/^File:/i, '') });
            }
          } catch (_) {}
        }

        if (collected.length < 4 && (scientificName || englishName)) {
          try {
            const wikiTitle = encodeURIComponent((scientificName || englishName).trim().replace(/\s+/g, '_'));
            const wikiRes = await fetch(`https://en.wikipedia.org/api/rest_v1/page/summary/${wikiTitle}`);
            if (wikiRes.ok) {
              const wikiJson = await wikiRes.json();
              const thumb = wikiJson?.thumbnail?.source || wikiJson?.originalimage?.source;
              if (thumb && !seenUrls.has(thumb)) {
                seenUrls.add(thumb);
                collected.push({ src: thumb, title: wikiJson.title || scientificName });
              }
            }
          } catch (_) {}
        }

        const slots = [];
        for (let i = 0; i < 4; i++) slots.push(collected[i] || null);
        return slots;
      }

        ensureAudioContext();
        const apiKey = getApiKey();
        if (!apiKey) {
          showApiKeyModal((savedKey) => { if (savedKey) startIdentification(imagePack); });
          return;

        const loadingText = document.getElementById('loadingText');
        loadingText.textContent = '낙엽 밟으며 식물을 판정하는 중...';
        overlay.classList.add('active');
        playLeafRustleSound();
        try {
          const [dateText, locationText, plantInfo] = await Promise.all([
            state.settings.recordDate ? Promise.resolve(formatNowKorean()) : Promise.resolve(''),
            state.settings.recordLocation ? getCurrentLocationText() : Promise.resolve(''),

          loadingText.textContent = '대표 식물 사진과 비교하는 중...';

          state.view = 'result';

          overlay.classList.remove('active');
        }
      }

      function escapeHtml(str) {
        return String(str || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
      }

      function escapeAttr(str) {
        return escapeHtml(str).replace(/'/g, '&#39;');
      }

      async function handleFileChange(event) {
        const file = event.target.files && event.target.files[0];
        event.target.value = '';
        if (!file) return;
        try {
          const pack = await readImageFile(file);
          state.previewUrl = pack.dataUrl;
          state.previewBase64 = pack.base64;
          state.previewMimeType = pack.mimeType;
          render();
          await startIdentification(pack);
        } catch (_) {
          showAlertModal('오류', '이미지를 불러오는 중 문제가 발생했어요.');
        }
      }

      document.getElementById('cameraInput').addEventListener('change', handleFileChange);
      document.getElementById('albumInput').addEventListener('change', handleFileChange);
